cmake_minimum_required(VERSION 3.24)
project(NexaExplorerCommand LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Explorer integration is Windows-only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(Nexa.ExplorerCommand SHARED ExplorerCommand.cpp exports.def)
target_compile_features(Nexa.ExplorerCommand PRIVATE cxx_std_20)
target_compile_definitions(Nexa.ExplorerCommand PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(Nexa.ExplorerCommand PRIVATE /W4 /WX /utf-8 /guard:cf)
target_link_options(Nexa.ExplorerCommand PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(Nexa.ExplorerCommand PRIVATE shell32 shlwapi ole32)
enable_testing()
add_executable(Nexa.ShellRegistration WIN32 Registration.cpp)
target_compile_features(Nexa.ShellRegistration PRIVATE cxx_std_20)
target_compile_definitions(Nexa.ShellRegistration PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(Nexa.ShellRegistration PRIVATE /W4 /WX /utf-8 /guard:cf)
target_link_options(Nexa.ShellRegistration PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(Nexa.ShellRegistration PRIVATE windowsapp shell32 shlwapi crypt32 advapi32)
if(NEXA_SHELL_CERT_SHA256)
  if(NOT NEXA_SHELL_CERT_SHA256 MATCHES "^[A-Fa-f0-9]+$")
    message(FATAL_ERROR "Invalid certificate fingerprint")
  endif()
  target_compile_definitions(Nexa.ShellRegistration PRIVATE NEXA_SHELL_CERT_SHA256="${NEXA_SHELL_CERT_SHA256}")
endif()
add_executable(ExplorerCommandTests Tests.cpp)
target_compile_features(ExplorerCommandTests PRIVATE cxx_std_20)
target_compile_definitions(ExplorerCommandTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ExplorerCommandTests PRIVATE /W4 /WX /utf-8)
target_link_libraries(ExplorerCommandTests PRIVATE shell32 shlwapi ole32)
add_test(NAME ExplorerCommandContracts COMMAND ExplorerCommandTests $<TARGET_FILE:Nexa.ExplorerCommand>)
add_executable(ShellTrustTests TrustTests.cpp)
target_compile_features(ShellTrustTests PRIVATE cxx_std_20)
target_compile_definitions(ShellTrustTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ShellTrustTests PRIVATE /W4 /WX /utf-8)
target_link_libraries(ShellTrustTests PRIVATE windowsapp crypt32 advapi32)
add_test(NAME ShellTrustOwnership COMMAND ShellTrustTests "${CMAKE_CURRENT_SOURCE_DIR}/test-signer.cer")
